 K&=j+1-p,\quad E=1+B(HT+p-1),\quad W_0=(1-H)j+1-p,\notag\\

 +TB[Hj+j+1+B(Hj(2j+2-p)+p(2j+1-p))]\notag\\

       +B^2HT^2j(Hj-j+p)(1+Bp),\notag\\
 F_1&=-2BHjp+BHj+2Bjp-2Bp^2+2Bp-Hj-Hp-H+j+1-p^2/j+p/j,\notag\\
