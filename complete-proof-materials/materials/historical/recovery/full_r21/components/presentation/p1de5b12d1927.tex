\section{The doubly infinitesimal endpoint with bounded $\beta/q^2$}
\label{p1de5b12d1927:sec:new-c144-small-nu-bounded-beta}

\paragraph{Scope.}
Keep the original entered physical C14.4 source, including its strict
gaps, nonnegative weights and prior-complement intersections. In this
section
\[
 \begin{gathered}
 S=\sigma+L\in\operatorname{Inf}_{>0},\quad
 \kappa=x/S\in\operatorname{Inf}_{>0},\quad
 \nu=\sigma/S\in\operatorname{Inf}_{>0},\\
 \ell=L/S=1-\nu,\quad q=\nu/\ell,\quad
 n_s=\beta/q^2\ \hbox{is bounded}.
 \end{gathered}
 \tag{SN1}
\]
The conclusion is finite separated tensor SOS for the complete original
parent, all testing angles and all real or complex states. In particular
no bound is imposed on $W/Z$, $u/w$, or any ratio of a testing
infinitesimal to a source infinitesimal. This is a source subdomain;
$\beta/q^2$ unbounded is not claimed here.

\paragraph{Opening-safe source bounds.}
Use the exact native source dictionary FM2--FM3 and the surviving
SE1--SE3 sphere identities. Put
\[
 c_0=SC(1-r),\quad a_0=SAr,\quad d_s=D/(Sq^2),\quad
 a_1=\alpha/S,\quad \mu=m/\kappa^2,\quad
 e=\kappa h/\nu,\quad \tau=h/q.
 \tag{SN2}
\]
The sphere gives, with its original nonnegative coordinates $d_0,b_0$,
\[
 d_s=\frac{\ell^2\mathcal H d_0^2}{2(P/S)J},\qquad
 n_s=\frac{\ell^2\mathcal H b_0^2}{2\kappa(P/S)\theta},\qquad
 Jd_s+\kappa\theta n_s=\frac{\ell^2\mathcal H}{2(P/S)}.
 \tag{SN3}
\]
Here $P/S,J,B/S,H_0,\mathcal H$ tend to $1$ on SN1, using SE4--SE6.
Thus $b_0^2=O(\kappa\theta n_s)$ is infinitesimal and $d_s\sim1/2$.
The exact identity $e(B/S)=\kappa H_0+\nu\mathcal H b_0^2$ now yields
$e/\kappa\sim1$. Consequently
\[
 h\sim\nu,\quad\theta\sim1,\quad\tau\sim1,\quad
 \mu\sim\tfrac12,\quad c_0,a_0,a_1\sim\tfrac12,
 \qquad D,\alpha,\beta,m\in\operatorname{Inf}_{\ge0}.
 \tag{SN4}
\]
All normalized coefficients in SN2 are bounded; their positive pivots
have real lower bounds independent of $r,1-r$. Literal $\beta=0$ is
included. Literal $D=0$ has empty intersection with SN1 by SN3. No
comparison of $S,q,\kappa$ is needed.

\paragraph{The complete reciprocal parent.}
For infinitesimal Alice coordinate $u$ and reciprocal Bob coordinate

\[
\begin{split}

\end{split}\tag{SN5}
\]
Keep both unrestricted state normals
\[
 X_e=X+ewW,\qquad N_y=(Y-ruW)/\sqrt{Sr(1-r)}.
 \tag{SN6}
\]
Every inverse and radical in these expressions is pure-source or

\begin{gathered}

\paragraph{Two free helper pivots.}
The exact polynomial identity
\[

       +2\kappa\ell^2w^2B^2-\ell^2(wB_u)^2

\paragraph{Exact scalar dictionary and the $1/\kappa$ cancellations.}
Put $d=D+1/2$ and define the following polynomials in $a,c$:
\[
\begin{split}
 A_P={}&(1+c)\{c_0(1-r)+a_0r(1+a)\}
       +S\{d(1-r-e)^2c+(\alpha+\tfrac12)(1+a)r^2\},\\
 A_N={}&m(1+ra+ec)^2+(\beta+\tfrac12)e^2c(1+c),\\
 K={}&m(1+ra+ec)+d(1-r-e)
       +(\alpha+\tfrac12)(1+a)r+(\beta+\tfrac12)e(1+c),\\
 F_Z={}&D+\alpha c+\beta a+(m+\alpha+\beta+\tfrac12)ac,\\
 J_1={}&K-\ell A_P,\quad L_1=J_1-(\ell/\kappa)A_N,\\
 H={}&F_Z-\ell^2ScA_P
                 +ac\{(\ell/\kappa)^2A_N-2(\ell/\kappa)J_1\}.
\end{split}\tag{SN9}
\]
These are the exact graph coefficients at $X=-ewW,Y=ruW$;
the negative $dac,e\{2(1-r)-e\}$ contribution has been placed
in $A_P$, not left in $A_N$. In particular $A_N/\kappa^2$ has
bounded polynomial coefficients and residue $1/2$.

The actual source satisfies

\[
\begin{gathered}

        +(a-x)\delta(-qn_s+cH_{11})+c(a-x)^2H_{20}.
\end{gathered}\tag{SN10}
\]
The quotients mean literal polynomial differences at $a=x,c=q$,
not division by a mixed tensor. A fully explicit implementation is
\path{c144_small_nu_dictionary.py}. For example $L_a$ is
$(L_1(a,c)-L_1(x,c))/(a-x)$, and $L_c$ is the quotient of
$L_1(x,c)-L_1(x,q)$ by $c-q$.

Individually $H_{20},H_{11},L_c/\kappa$ need not be bounded.

 \tag{SN16}

\begin{gathered}
