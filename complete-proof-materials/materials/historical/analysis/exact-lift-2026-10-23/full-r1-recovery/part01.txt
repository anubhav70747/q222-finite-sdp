import json

import sys

dest=Path(sys.argv[1]).resolve()
for name in sys.argv[2:]:
    text=Path(name).read_text()
    pos=0
    while pos<len(text):
        end=text.index('\n',pos)
        line=text[pos:end]; pos=end+1
